📝 Đoạn mã Google Apps Script doPost(e) đồng bộ vào Google Sheets: Để Cột K (Tin tuyển dụng gốc (File/Ảnh/Link)) và các cột khác nhận chính xác dữ liệu từ form web, trong Extensions (Tiện ích mở rộng) ➔ Apps Script của Google Sheet, bạn chỉ cần dán hàm sau: JavaScript function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Theo dõi ứng tuyển"); // Tìm dòng trống tiếp theo var lastRow = sheet.getLastRow() + 1; var stt = lastRow - 6; // Bắt đầu tính STT từ dòng 7 // Xử lý link tin gốc (Cột K) var originalLinkFormula = data.original_jd_link ? '=HYPERLINK("' + data.original_jd_link + '", "🔗 Link tin tuyển dụng")' : "Không có link"; // Xử lý link CV (Cột L) var cvFormula = data.cv_link ? '=HYPERLINK("' + data.cv_link + '", "📄 Xem CV")' : "Chưa đính kèm"; // Ghi các cột từ B đến M: // [STT, Tên công ty, Vị trí, Mức lương, Thời gian, Yêu cầu, Tóm tắt Gemini, Địa điểm, Liên hệ, Tin gốc, CV, Trạng thái] var rowData = [ stt, data.company_name || "", data.job_title || "", data.salary || "Thỏa thuận", data.working_time || "Giờ hành chính", data.requirements || "", data.jd_summary || "", data.location || "", data.contact || "", originalLinkFormula, cvFormula, data.status || "Chưa gửi CV" ]; sheet.getRange(lastRow, 2, 1, rowData.length).setValues([rowData]); return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); }
